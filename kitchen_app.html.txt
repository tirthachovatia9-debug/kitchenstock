<script>
let inventory = JSON.parse(localStorage.getItem("inventory")) || [];
let expenses = JSON.parse(localStorage.getItem("expenses")) || [];

function save() {
    localStorage.setItem("inventory", JSON.stringify(inventory));
    localStorage.setItem("expenses", JSON.stringify(expenses));
}

function addItem() {
    let item = document.getElementById("item").value;
    let qty = parseInt(document.getElementById("qty").value);
    let min = parseInt(document.getElementById("min").value);

    inventory.push({item, qty, min});
    save();
    render();
}

function addExpense() {
    let cat = document.getElementById("category").value;
    let amt = parseInt(document.getElementById("amount").value);

    expenses.push({cat, amt});
    save();
    render();
}

function render() {
    let invHTML = "";
    let lowHTML = "";

    inventory.forEach((i, index) => {
        let low = i.qty <= i.min;
        invHTML += `
        <div class="card">
            ${i.item} - ${i.qty}
            ${low ? '<span class="low"> LOW</span>' : ''}
        </div>`;

        if (low) {
            lowHTML += `
            <div class="card low">
                ${i.item} - ${i.qty}
            </div>`;
        }
    });

    document.getElementById("inventoryList").innerHTML = invHTML;
    document.getElementById("lowList").innerHTML = lowHTML;

    let expHTML = "";
    let total = 0;

    expenses.forEach(e => {
        total += e.amt;
        expHTML += `
        <div class="card">
            ${e.cat} - ₹${e.amt}
        </div>`;
    });

    document.getElementById("expenseList").innerHTML = expHTML;
    document.getElementById("total").innerText = "Total: ₹" + total;
}

function show(tab) {
    document.getElementById("inventory").classList.add("hidden");
    document.getElementById("lowStock").classList.add("hidden");
    document.getElementById("expense").classList.add("hidden");

    document.getElementById(tab).classList.remove("hidden");
}

render();
</script>
